 131    def fit_opti(self, X, X_dot, C=100, deltas=0.01, norm='L2', segmentation='strict', segments=[], print_on=False, apply_LR=False):
 132        '''
 133        Fit the Hybrid Regressor model using the *optimization based* approach.
 134
 135        Attributes
 136        ----------
 137        X : ndarray (N x D)
 138            A matrix of training data points [x1, x2, ..., xN].
 139        
 140        X_dot : ndarray (N x D)
 141            A matrix of corresponding estimated derivatives of training data points [x1dot, x2dot, ..., xNdot]
 142        
 143        C : float, default=100
 144            The hyperparameter C controlling the strength of regularization in the optimization objective function.
 145
 146        detlas : list of `n` floats or float, default=0.01
 147            A list of absolute insensitivity errors for each mode.
 148            When one value is given, the same value applies to all modes.
 149
 150        norm : 'L1' or 'L2', default='L2'
 151            The norm used for regularization.
 152            Note: 'L1' norm is not supported (may or may not work).
 153
 154        segmentation : None or 'strict' or 'squared_diff', default='strict'
 155            The type of integration of segmentation information.
 156            None means no segmentation information is integrated.
 157            'strict' means segmentation will be strictly integrated, and Xi variables per mode per point will be replaced by Xi variables per mode per segment. This is the method illustrated in the thesis.
 158            'squared_diff' means segmentation will be added by adding the squared difference of Xi variables of points belonging to the same segment to the minimization problem. This is not supported by expriments.
 159
 160        segments : ndarray of N ints, default=[]
 161            The segment MAPPING of each data point to its segment.
 162            segments[i] indicates the segment identifier of point X[i].
 163            This is DIFFERENT from `segments` in `fit_first_opti` which is a LIST OF CHANGE POINT INDICES.
 164
 165        print_on : Bool, default=False
 166            The verbosity parameter of the Casadi optimizer.
 167
 168        apply_LR : Bool, default=False
 169            Indicates whether to apply linear/polynomial regression on data after finding their modes.
 170            This replaces the continuous dynamics found by the optimizer.
 171        '''
 172
 173
 174        if self.n is None:
 175            raise ValueError("To use the optimization based method, the number of modes n must be provided.")
 176        if not segmentation is None and segmentation not in ['squared_diff', 'strict']:
 177            raise ValueError("The parameter segmentation must be one of {None, 'squared_diff', 'strict'}")
 178        if not segmentation is None and len(segments) == 0:
 179            raise ValueError("The parameter segments must be provided or non-empty when a segmentation integration method is selected.")
 180
 181        tc = self.train_coefs
 182        n = self.n
 183        deg = self.deg
 184        if not tc:
 185            kernel_mat_list = []
 186            for j in range(n):
 187                if isinstance(self.kernels_list[j], str):
 188                    kernel = kernel_dic[self.kernels_list[j]]
 189                else:
 190                    kernel = self.kernels_list[j]
 191                
 192                kernel_mat_list.append(kernel(X, X, self.kargs_list[j]))
 193
 194        if tc:
 195            poly = PolynomialFeatures(degree=deg, include_bias=False)
 196            X_trans = poly.fit_transform(X)
 197            self.poly = poly
 198        else:
 199            X_trans = X
 200
 201        # deltas could be a scalar or defined per mode
 202        if (np.isscalar(deltas)):
 203            temp = np.ones(n)
 204            deltas = deltas*temp
 205        else:
 206            assert(len(deltas) == n)
 207
 208        N = X_dot.shape[0]  # nb of points
 209        D = X_dot.shape[1]  # dimension of each vector
 210        Q = X_trans.shape[1]
 211        S = max(segments) + 1   # Number of segments
 212        opti = casadi.Opti()
 213
 214        # variable declaration arrays
 215        if tc:
 216            Omegas = np.ndarray((n, D), dtype=object)
 217        else:
 218            Alphas = np.ndarray((n, D), dtype=object)
 219        betas = np.ndarray((n, D), dtype=object)
 220        Xis = np.ndarray((n,), dtype=object)
 221
 222        if norm == 'L1':
 223            As = np.ndarray((n, D), dtype=object)
 224        
 225        for j in range(n):
 226            if segmentation == 'strict':
 227                xi_size_per_mode = S
 228            else:
 229                xi_size_per_mode = N
 230            
 231            Xis[j] = opti.variable(xi_size_per_mode)
 232            opti.set_initial(Xis[j], np.random.rand(xi_size_per_mode))
 233
 234            for d in range(D):
 235                if tc:
 236                    Omegas[j][d] = opti.variable(Q)
 237                    opti.set_initial(Omegas[j][d], np.random.rand(Q))
 238                else:
 239                    Alphas[j][d] = opti.variable(N)
 240                    opti.set_initial(Alphas[j][d], np.random.rand(N))
 241
 242                betas[j][d] = opti.variable()
 243                opti.set_initial(betas[j][d], np.random.rand())
 244                #betas[j][d] = 0
 245                if norm == 'L1':
 246                    if not tc:
 247                        As[j][d] = opti.variable(N)
 248                        opti.set_initial(As[j][d], np.random.rand(N))
 249                    else:
 250                        As[j][d] = opti.variable(Q)
 251                        opti.set_initial(As[j][d], np.random.rand(Q))
 252
 253        C_param = opti.parameter()
 254        opti.set_value(C_param, C)
 255
 256        # First part of the optimization objective function:
 257        # the sum of products of Xis
 258        # This forces one Xi per data point to be ~0, which dictates the mode
 259
 260        sum_products_xis = 0
 261        for i in range(xi_size_per_mode):
 262            temp = 1
 263            for j in range(n):
 264                temp *= Xis[j][i]
 265            sum_products_xis += temp
 266
 267        #sum_products_xis *= C_param
 268
 269        # Second part of the optimization objective function:
 270        # The sum of norms of alphas/Omegas
 271        # Here we use the L1 norm
 272        if norm == 'L1':
 273            if not tc:
 274                eye = np.ones(N)
 275            else:
 276                eye = np.ones(Q)
 277            sum_norms_a = 0
 278            for j in range(n):
 279                for d in range(D):
 280                    sum_norms_a += (As[j][d].T @ eye)
 281            to_minimize = sum_norms_a + sum_products_xis
 282        elif norm == 'L2':
 283            sum_norms = 0
 284            for j in range(n):
 285                for d in range(D):
 286                    if tc:
 287                        sum_norms += (Omegas[j][d].T @ Omegas[j][d])
 288                    else:
 289                        sum_norms += (Alphas[j][d].T @ Alphas[j][d])
 290            to_minimize = C_param*sum_norms + sum_products_xis
 291        else:
 292            raise Exception("The norm parameter has to be 'L1' or 'L2'.")
 293        
 294
 295        if segmentation == 'squared_diff':
 296            sum_diff_xis = 0
 297            for r in range(S):
 298                seg_indices = np.where(segments==r)
 299                for i in range(seg_indices.shape[0] - 1):
 300                    for j in range(n):
 301                        sum_diff_xis += (Xis[j][seg_indices[i]] - Xis[j][seg_indices[i+1]])**2
 302            # for r in range(len(segments)-1):
 303            #     for i in range(segments[r], segments[r+1] - 1):
 304            #         for j in range(n):
 305            #             sum_diff_xis += (Xis[j][i] - Xis[j][i+1])**2
 306            # for i in range(segments[-1], N-1):
 307            #     for j in range(n):
 308            #         sum_diff_xis += (Xis[j][i] - Xis[j][i+1])**2
 309            
 310            to_minimize += C*sum_diff_xis
 311
 312        # Setting the objective function
 313        opti.minimize(to_minimize)
 314
 315        # CONSTRAINTS
 316
 317        # all xis must be positive
 318        for j in range(n):
 319            opti.subject_to(Xis[j] >= 0)
 320        
 321        x_dotT = X_dot.T    # d by N
 322        eye = np.ones(N)
 323        for d in range(D):
 324            for j in range(n):
 325                if segmentation == 'strict':
 326                    # # Here we need to transform each Xi[j], which is currently an S by 1 matrix into an N by 1 matrix
 327                    # # We do this by constructing a transformation matrix that assigns the correct xi according to which segment the point is in
 328                    # trans_mat = np.zeros((N, S))
 329                    # for i in range(N):
 330                    #     seg_index = which_segment(i, segments)
 331                    #     trans_mat[i][seg_index] = 1
 332                    # Xis[j] = casadi.vec(trans_mat @ Xis[j])
 333                    for i in range(N):
 334                        seg_index = segments[i]
 335                        if not tc:
 336                            opti.subject_to(x_dotT[d][i] - casadi.dot(kernel_mat_list[j][i].T, Alphas[j][d]) - betas[j][d] <=  deltas[j] + Xis[j][seg_index])
 337                            opti.subject_to(x_dotT[d][i] - casadi.dot(kernel_mat_list[j][i].T, Alphas[j][d]) - betas[j][d] >= -deltas[j] - Xis[j][seg_index])
 338                        else:
 339                            opti.subject_to(x_dotT[d][i] - casadi.dot(X_trans[i].T, Omegas[j][d]) - betas[j][d] <=  deltas[j] + Xis[j][seg_index])
 340                            opti.subject_to(x_dotT[d][i] - casadi.dot(X_trans[i].T, Omegas[j][d]) - betas[j][d] >= -deltas[j] - Xis[j][seg_index])
 341                else:
 342                    # the difference between prediction and the ground truth is bounded in [-delta-xi, delta+xi]
 343                    if not tc:
 344                        opti.subject_to(x_dotT[d] - casadi.vec(kernel_mat_list[j] @ Alphas[j][d]) - betas[j][d]*eye <=  deltas[j]*eye + Xis[j])
 345                        opti.subject_to(x_dotT[d] - casadi.vec(kernel_mat_list[j] @ Alphas[j][d]) - betas[j][d]*eye >= -deltas[j]*eye - Xis[j])
 346                    else:
 347                        opti.subject_to(x_dotT[d] - casadi.vec(X_trans @ Omegas[j][d]) - betas[j][d]*eye <=  deltas[j]*eye + Xis[j])
 348                        opti.subject_to(x_dotT[d] - casadi.vec(X_trans @ Omegas[j][d]) - betas[j][d]*eye >= -deltas[j]*eye - Xis[j])
 349                if norm=='L1':
 350                    if not tc:
 351                        # As are bounded by [-alpha, +alpha]
 352                        opti.subject_to(-Alphas[j][d] <= As[j][d])
 353                        opti.subject_to(As[j][d] <= Alphas[j][d])
 354                    else:
 355                        # As are bounded by [-omega, +omega]
 356                        opti.subject_to(-Omegas[j][d] <= As[j][d])
 357                        opti.subject_to(As[j][d] <= Omegas[j][d])
 358
 359        opti.solver('ipopt', dict(print_time = 0), dict(max_iter = 3000, print_level = 0))
 360        # Initiate the optimization
 361        sol = opti.solve()
 362
 363        # Printing the results if print is on
 364        if print_on:
 365            for j in range(n):
 366                print("Xi_" + str(j) + ": " + str(sol.value(Xis[j])))
 367            for j in range(n):
 368                for d in range(D):
 369                    if not tc:
 370                        print("Alpha_" + str(j) + "_" + str(d) + ": " + str(sol.value(Alphas[j][d])))
 371                    else:
 372                        print("Omega_" + str(j) + "_" + str(d) + ": " + str(sol.value(Omegas[j][d])))
 373                    print( "Beta_" + str(j) + "_" + str(d) + ": " + str(sol.value(betas[j][d])))
 374
 375        # Saving the results in proper arrays
 376        if not tc:
 377            Alphas_sol = np.ndarray((n, D, N))
 378        else:
 379            Omegas_sol = np.ndarray((n, D, Q))
 380        betas_sol  = np.ndarray((n, D))
 381        Xis_sol   = np.ndarray((n, xi_size_per_mode))
 382
 383        for j in range(n):
 384                Xis_sol[j] = sol.value(Xis[j])
 385        for j in range(n):
 386            for d in range(D):
 387                if not tc:
 388                    Alphas_sol[j][d] = sol.value(Alphas[j][d])
 389                else:
 390                    Omegas_sol[j][d] = sol.value(Omegas[j][d])
 391                betas_sol[j][d]  = sol.value(betas[j][d])
 392        
 393        if segmentation == 'strict':
 394            self.Xis_segments_ = Xis_sol
 395            final_Xis = np.ndarray((n, N))
 396            for j in range(n):
 397                for i in range(N):
 398                    seg_index = segments[i]
 399                    final_Xis[j][i] = Xis_sol[j][seg_index]
 400
 401            self.Xis_ = final_Xis
 402        else: 
 403            self.Xis_ = Xis_sol
 404
 405        if apply_LR and tc:
 406            modes = self.get_modes()
 407            LR = LinearRegression()
 408            for j in range(n):
 409                indices_j = np.where(modes == j)
 410                X_j = X_trans[indices_j]
 411                X_dot_j = X_dot[indices_j]
 412                LR.fit(X_j, X_dot_j)
 413                Omegas_sol[j] = LR.coef_
 414                betas_sol[j] = LR.intercept_
 415
 416
 417
 418
 419        #self.sol_ = sol
 420        if not tc:
 421            self.Alphas_ = Alphas_sol
 422        else:
 423            self.Omegas_ = Omegas_sol
 424        self.betas_ = betas_sol
 425        # self.Xis_ = Xis_sol
 426        self.Xfit_ = X
 427        self.Xfit_trans_ = X_trans
 428        self.X_dot_fit_ = X_dot
 429
 430        self.modesfit_ = self.get_modes() # here we store the modes of each data point used for fitting

 536        RDs = np.ndarray((self.n,))
 537        for j in range(self.n):
 538            Xdot_pred = self.predict(X, lambda x: j)
 539            RDs[j] = np.mean(relative_diff(Xdot, Xdot_pred))
 540            # print("mode", j, RDs[j])
 541            # print(Xdot - Xdot_pred)
 542            # print(Xdot, Xdot_pred)
 543            # RDs[j] = np.linalg.norm(Xdot - Xdot_pred, ord='fro')
 544        min_RD = np.min(RDs)
 545        # print(RDs)
 546        if min_RD < tol:
 547            print("No need to improve, mode", np.argmin(RDs), f"matches. RD = {min_RD:.3e}")
 548        else:
 549            print(f"Need to improve, min(RDs) = {min_RD:.3e}")
 550            # print(RDs)
 551        return min_RD < tol, np.argmin(RDs)

 553    def update_continuous(self, X, X_dot, good_RD=0.001, regressor=LinearRegression(), Time = None):
 554        '''
 555        Update the learned continuous dynamics and mode information with NEW data of ONE SEGMENT.
 556        This function is called repeatedly on data of different segments in the incremental learning approach.
 557        Theoretically can be used to update the dynamics found using the optimization based appraoch, but this is not tested.
 558
 559        Attributes
 560        ----------
 561        X : ndarray (N x D)
 562            A matrix of NEW training data points [x1, x2, ..., xN] belonging in ONE SEGMENT.
 563        
 564        X_dot : ndarray (N x D)
 565            A matrix of corresponding estimated derivatives of training data points [x1dot, x2dot, ..., xNdot]
 566
